"""Gemini 3.8 Flash / High Runner (Published Antigravity Preflight Route)."""
import subprocess
import time
from pathlib import Path
from typing import Dict, Any

from runners.base_runner import BaseRunner

AGY_PATH = "<HOST_AGY_PATH>"


class GeminiFlashRunner(BaseRunner):
    def __init__(self):
        super().__init__(name="Gemini 3.8 Flash (High)", model_tag="gemini-3.8-flash-high")

    def run(self, case_path: Path) -> Dict[str, Any]:
        case = self.load_case(case_path)
        prompt = self.build_prompt(case)
        
        cmd = [
            AGY_PATH,
            "--model", self.model_tag,
            "--effort", "high",
            "--mode", "plan",
            "--dangerously-skip-permissions",
            "-p", prompt
        ]
        
        t0 = time.time()
        try:
            res = subprocess.run(
                cmd,
                capture_output=True,
                text=True,
                timeout=240,
                check=False
            )
            elapsed = time.time() - t0
            return {
                "runner": self.name,
                "model": self.model_tag,
                "case_id": case["id"],
                "exit_code": res.returncode,
                "elapsed_seconds": round(elapsed, 2),
                "stdout": res.stdout,
                "stderr": res.stderr,
                "error": None
            }
        except Exception as e:
            elapsed = time.time() - t0
            return {
                "runner": self.name,
                "model": self.model_tag,
                "case_id": case["id"],
                "exit_code": -1,
                "elapsed_seconds": round(elapsed, 2),
                "stdout": "",
                "stderr": str(e),
                "error": str(e)
            }
