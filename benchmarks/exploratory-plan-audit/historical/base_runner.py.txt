"""Base runner interface for Antigravity Preflight Benchmark."""
import json
import time
from pathlib import Path
from typing import Dict, Any


class BaseRunner:
    def __init__(self, name: str, model_tag: str):
        self.name = name
        self.model_tag = model_tag

    def load_case(self, case_path: Path) -> Dict[str, Any]:
        with open(case_path, 'r', encoding='utf-8') as f:
            return json.load(f)

    def build_prompt(self, case: Dict[str, Any]) -> str:
        prompt = (
            "You are an adversarial preflight auditor evaluating a candidate plan before execution.\n"
            "Your objective: Identify fatal tripwires that will cause late-stage failure (tooling assertions, "
            "transport/timeout overruns, contract violations).\n\n"
            f"### CANDIDATE PLAN: {case['candidate_plan']['title']}\n"
            f"{json.dumps(case['candidate_plan'], indent=2)}\n\n"
            "### EVALUATION INSTRUCTIONS:\n"
            "Do not invoke operational tools, execute code, or run shell commands. Perform this evaluation strictly as a read-only static text audit.\n"
            "1. Output a verdict: 'GO' (safe to proceed) or 'NO-GO' (fatal tripwire detected).\n"
            "2. If NO-GO, identify the exact fatal tripwire and the concrete failure mechanism.\n"
            "3. If GO, confirm compliance with contract limits and error-handling requirements.\n"
            "Keep your evaluation concise, factual, and strictly focused on execution safety.\n"
        )
        return prompt

    def run(self, case_path: Path) -> Dict[str, Any]:
        raise NotImplementedError
